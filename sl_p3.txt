function toast(msg, dur) {
  var t = document.getElementById('toast');

function fmt(n) { return Number(n || 0).toLocaleString('he-IL'); }

function todayStr() {
  var d = new Date();
  return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
}

function acadYearOf(dateStr) {
  var d = new Date(dateStr + 'T00:00:00');
  return d.getMonth() >= 8 ? d.getFullYear() : d.getFullYear() - 1;
}

  var m = {2020:'תש"פ',2021:'תשפ"א',2022:'תשפ"ב',2023:'תשפ"ג',2024:'תשפ"ד',2025:'תשפ"ה',2026:'תשפ"ו',2027:'תשפ"ז',2028:'תשפ"ח',2029:'תשפ"ט',2030:'תש"צ'};
  return (m[y] || y) + '–' + (m[y+1] || (y+1));

async function syncAll() {
  try {
    var results = await Promise.all([
      SB.from('sl_students').select('*').order('name'),
      SB.from('sl_transactions').select('*').order('date'),
      SB.from('sl_settings').select('*'),
      SB.from('sl_lists').select('*').order('value')
    ]);
    if (results[0].data) STUDENTS = results[0].data;
    if (results[1].data) TRANSACTIONS = results[1].data;
    if (results[2].data) {
      SETTINGS = {};
      results[2].data.forEach(function(r){ SETTINGS[r.key] = r.value; });
    }
    if (results[3].data) {
      LISTS = {};
      results[3].data.forEach(function(r){
        if (!LISTS[r.category]) LISTS[r.category] = [];
        LISTS[r.category].push({ id: r.id, value: r.value });
      });
    }
    refreshUI();
  } catch(e) { console.error('[sync]', e); }
}

  var grandTotal = chartTotals.reduce(function(a, b){ return a + b; }, 0);

  var calYear = month >= 8 ? year : year + 1;

// ===== DISTRIBUTION LOGIC =====
function calcDistribution(studentId, year) {
  var student = STUDENTS.find(function(s){ return s.id === studentId; });
  if (!student) return {};
  var defT = parseInt(SETTINGS['default_tuition'] || 2000);
  var cs = student.card_settings || {};
  var stuT = cs.monthly_tuition ? parseInt(cs.monthly_tuition) : defT;

  var txns = TRANSACTIONS.filter(function(t){
    return t.student_id === studentId && acadYearOf(t.date) === year;
  }).sort(function(a, b){ return a.date > b.date ? 1 : a.date < b.date ? -1 : 0; });

  var pool = txns.reduce(function(sum, t){ return sum + parseFloat(t.amount); }, 0);

  var result = {};
  YEAR_MONTHS.forEach(function(m){
    result[m] = { req: stuT, paid: 0 };
    if (pool <= 0) return;
    if (pool >= stuT) { result[m].paid = stuT; pool -= stuT; }
    else { result[m].paid = Math.round(pool * 100) / 100; pool = 0; }
  });
  return result;
}

async function saveTxn() {
  var stId = parseInt(document.getElementById('sd-val-student').value);
  var date = document.getElementById('txn-date').value;
  var amount = parseFloat(document.getElementById('txn-amount').value);
  var method = document.getElementById('txn-method').value;
  var note = document.getElementById('txn-note').value.trim();

  if (!stId) { toast('נא לבחור תלמיד'); return; }
  if (!date) { toast('נא לבחור תאריך'); return; }
  if (!amount || amount <= 0) { toast('נא להזין סכום תקין'); return; }

  try {
    var r = await SB.from('sl_transactions').insert({
      student_id: stId, date: date, amount: amount,
      payment_method: method || null, note: note || null
    });
    if (r.error) throw r.error;
    toast('תשלום נשמר ✓');
    document.getElementById('txn-amount').value = SETTINGS['default_tuition'] || '2000';
    document.getElementById('txn-note').value = '';
    await syncAll();
  } catch(e) { toast('שגיאה: ' + (e.message || e)); }
}

function renderTxnLog() {
  var filterId = parseInt(document.getElementById('sd-val-txnfilter').value) || 0;
  var txns = TRANSACTIONS.slice().sort(function(a, b){ return a.date > b.date ? -1 : 1; }).slice(0, 60);
  if (filterId) txns = txns.filter(function(t){ return t.student_id === filterId; });

  var html = '';
  txns.forEach(function(t){
    var st = STUDENTS.find(function(s){ return s.id === t.student_id; });
    var stName = st ? st.name : '#' + t.student_id;
    html += '<div class="txn-row">';
    html += '<span style="color:var(--text3);font-size:.78rem">' + t.date + '</span>';
    html += '<span style="font-weight:600">' + stName + '</span>';
    html += '<span style="font-weight:700;color:var(--green)">₪' + fmt(t.amount) + '</span>';
    html += '<span class="pill">' + (t.payment_method || '—') + '</span>';
    html += '<button class="btn sm danger" onclick="deleteTxn(' + t.id + ')">✕</button>';
    html += '</div>';
  });
  document.getElementById('txn-log').innerHTML = html || '<div class="empty">אין תשלומים</div>';
}

  var yearTxns = TRANSACTIONS.filter(function(t){
    return t.student_id === SC_STUDENT_ID && acadYearOf(t.date) === SC_YEAR;
  }).sort(function(a, b){ return a.date > b.date ? -1 : 1; });

  var pass = document.getElementById('set-pass-input').value;
  var adminPass = SETTINGS['admin_pass'] || 'admin';
  if (pass !== adminPass) { document.getElementById('set-pass-err').textContent = 'סיסמה שגויה'; return; }
  _settingsUnlocked = true;
  document.getElementById('settings-lock').style.display = 'none';
  document.getElementById('settings-main').style.display = 'block';
  document.getElementById('set-def-tuition').value = SETTINGS['default_tuition'] || '2000';
  document.getElementById('set-pass-err').textContent = '';
  renderSettingsLists();
}

async function saveDefaultTuition() {
  var val = document.getElementById('set-def-tuition').value;
  if (!val || isNaN(val)) { toast('ערך לא תקין'); return; }
  var r = await SB.from('sl_settings').upsert({ key: 'default_tuition', value: val });
  if (r.error) { toast('שגיאה: ' + r.error.message); return; }
  toast('נשמר ✓');
  await syncAll();
}

async function changeAdminPass() {
  var p1 = document.getElementById('set-new-pass').value;
  var p2 = document.getElementById('set-new-pass2').value;
  if (!p1) { toast('נא להזין סיסמה'); return; }
  if (p1 !== p2) { toast('הסיסמאות אינן תואמות'); return; }
  var r = await SB.from('sl_settings').upsert({ key: 'admin_pass', value: p1 });
  if (r.error) { toast('שגיאה: ' + r.error.message); return; }
  toast('סיסמה שונתה ✓');
  document.getElementById('set-new-pass').value = '';
  document.getElementById('set-new-pass2').value = '';
  await syncAll();
}

function renderSettingsLists() {
  var stHtml = STUDENTS.map(function(s){
    return '<div class="set-list-item">' +
      '<span style="font-weight:600">' + s.name + '</span>' +
      '<span class="badge ' + (s.active ? 'green' : 'gray') + '">' + (s.active ? 'פעיל' : 'לא פעיל') + '</span>' +
      '<button class="btn sm danger" onclick="deleteStudent(' + s.id + ')">✕ מחק</button>' +
      '</div>';
  }).join('');
  document.getElementById('set-students-list').innerHTML = stHtml || '<div class="empty">אין תלמידים</div>';
  renderListItems('payment_methods', 'set-methods-list');
  renderListItems('sections', 'set-sections-list');
}

function renderListItems(cat, elId) {
  var items = LISTS[cat] || [];
  var html = items.map(function(item){
    return '<div class="set-list-item">' +
      '<span>' + item.value + '</span>' +
      '<button class="btn sm danger" onclick="deleteListItem(' + item.id + ')">✕</button>' +
      '</div>';
  }).join('');
  document.getElementById(elId).innerHTML = html || '<div class="empty">ריק</div>';
}

async function addListItem(cat) {
  var val = prompt('הזן ערך חדש:');
  if (!val || !val.trim()) return;
  var r = await SB.from('sl_lists').insert({ category: cat, value: val.trim() });
  if (r.error) { toast('שגיאה: ' + r.error.message); return; }
  toast('נוסף ✓');
  await syncAll();
}

async function deleteListItem(id) {
  showConfirm('מחיקה', 'למחוק פריט זה?', async function(){
    var r = await SB.from('sl_lists').delete().eq('id', id);
    if (r.error) { toast('שגיאה: ' + r.error.message); return; }
    toast('נמחק ✓');
    await syncAll();
  });
}

function openAddStudent() {
  var secSel = document.getElementById('new-st-section');
  var sections = LISTS['sections'] || [];
  secSel.innerHTML = '<option value="">-- ללא --</option>';
  sections.forEach(function(s){ secSel.innerHTML += '<option value="' + s.value + '">' + s.value + '</option>'; });
  document.getElementById('new-st-name').value = '';
  document.getElementById('new-st-tuition').value = '';
  openModal('modal-add-student');
}

async function saveNewStudent() {
  var name = document.getElementById('new-st-name').value.trim();
  var tuition = document.getElementById('new-st-tuition').value;
  var section = document.getElementById('new-st-section').value;
  if (!name) { toast('שם חובה'); return; }
  var cs = {};
  if (tuition) cs.monthly_tuition = parseInt(tuition);
  if (section) cs.section = section;
  var r = await SB.from('sl_students').insert({ name: name, active: true, card_settings: cs });
  if (r.error) { toast('שגיאה: ' + r.error.message); return; }
  closeModal('modal-add-student');
  toast('תלמיד נוסף ✓');
  await syncAll();
}

async function deleteStudent(id) {
  var s = STUDENTS.find(function(x){ return x.id === id; });
  showConfirm('מחיקת תלמיד', 'למחוק את ' + (s ? s.name : 'תלמיד') + '?\nכל התשלומים שלו יימחקו גם כן.', async function(){
    var r = await SB.from('sl_students').delete().eq('id', id);
    if (r.error) { toast('שגיאה: ' + r.error.message); return; }
    if (SC_STUDENT_ID === id) { SC_STUDENT_ID = null; document.getElementById('sc-main').style.display = 'none'; document.getElementById('sd-input-sc').value = ''; document.getElementById('sd-val-sc').value = ''; }
    toast('תלמיד נמחק');
    await syncAll();
  });
}

function showConfirm(title, msg, cb) {
  document.getElementById('modal-confirm-title').textContent = title;
  document.getElementById('modal-confirm-msg').textContent = msg;
  _confirmCb = cb;
  document.getElementById('modal-confirm-ok').onclick = function(){
    closeModal('modal-confirm');
    if (_confirmCb) _confirmCb();
  };
  openModal('modal-confirm');
}
