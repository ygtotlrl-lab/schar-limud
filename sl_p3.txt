function calcDistribution(sid,year){
  var s=STUDENTS.find(function(x){return x.id===sid;});if(!s)return{};
  var cs=s.card_settings||{},defT=parseInt(SETTINGS['default_tuition']||2000),stuT=cs.monthly_tuition?parseInt(cs.monthly_tuition):defT;
  var txns=TRANSACTIONS.filter(function(t){return t.student_id===sid&&acadYearOf(t.date)===year;}).sort(function(a,b){return a.date>b.date?1:-1;});
  var pool=txns.reduce(function(s,t){return s+parseFloat(t.amount);},0),result={};
  YEAR_MONTHS.forEach(function(m){result[m]={req:stuT,paid:0};if(pool<=0)return;if(pool>=stuT){result[m].paid=stuT;pool-=stuT;}else{result[m].paid=Math.round(pool*100)/100;pool=0;}});
  return result;
}
